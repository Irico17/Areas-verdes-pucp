# Arnés de verificación automatizada para las tres rutas de baja lógica y PATCH de zona.
# Exclusivo para el backend nuevo (apps/api no registra estas rutas).
# Directiva: solo-nueva <rol> <METODO> <ruta> [@cuerpo.json] <status_esperado> [fragmento_esperado]
# Si fragmento_esperado inicia con !, se valida que el cuerpo NO contenga ese texto.

# 1. Rutas con recursos inexistentes (404)
solo-nueva admin POST /api/v1/catastro/areas/AV-INEXISTENTE-9999/baja 404 no encontrada
solo-nueva admin POST /api/v1/catastro/zonas-supervision/Z4/baja 404 no encontrada
solo-nueva admin PATCH /api/v1/catastro/zonas-supervision/Z4 @scripts/cuerpos/zona-patch-valido.json 404 no encontrada

# 2. Sin sesión (401)
solo-nueva anon POST /api/v1/catastro/areas/AV-BAJA-AUTO/baja 401 inicie sesión
solo-nueva anon POST /api/v1/catastro/zonas-supervision/Z4/baja 401 inicie sesión
solo-nueva anon PATCH /api/v1/catastro/zonas-supervision/Z4 @scripts/cuerpos/zona-patch-valido.json 401 inicie sesión

# 3. Rol sin permiso registrar: jefatura (403)
solo-nueva jefatura POST /api/v1/catastro/areas/AV-BAJA-AUTO/baja 403 no tiene ese permiso
solo-nueva jefatura POST /api/v1/catastro/zonas-supervision/Z4/baja 403 no tiene ese permiso
solo-nueva jefatura PATCH /api/v1/catastro/zonas-supervision/Z4 @scripts/cuerpos/zona-patch-valido.json 403 no tiene ese permiso

# 4. Creación de área y zona ficticias para las pruebas (201)
solo-nueva admin POST /api/v1/catastro/areas @scripts/cuerpos/area-crear-baja.json 201 AV-BAJA-AUTO
solo-nueva admin POST /api/v1/catastro/zonas-supervision @scripts/cuerpos/zona-crear-baja.json 201 Z4

# 5. Código del cuerpo distinto al de la ruta en PATCH zona (400)
solo-nueva admin PATCH /api/v1/catastro/zonas-supervision/Z4 @scripts/cuerpos/zona-patch-mismatch.json 400 no coincide con el cuerpo

# 6. PATCH de zona exitoso (nombre y area_m2) por coordinación (200)
solo-nueva coordinacion PATCH /api/v1/catastro/zonas-supervision/Z4 @scripts/cuerpos/zona-patch-valido.json 200 Zona Cuatro Modificada

# 7. Bajas lógicas exitosas con activo:false por coordinacion y admin (200)
solo-nueva coordinacion POST /api/v1/catastro/areas/AV-BAJA-AUTO/baja 200 "activo":false
solo-nueva admin POST /api/v1/catastro/zonas-supervision/Z4/baja 200 "activo":false

# 8. Bajas repetidas: idempotentes, 200 y sin duplicar historial
solo-nueva admin POST /api/v1/catastro/areas/AV-BAJA-AUTO/baja 200 "activo":false
solo-nueva admin POST /api/v1/catastro/zonas-supervision/Z4/baja 200 "activo":false

# 9. Tras la baja, el área ya no figura en /catastro/areas ni en /geo/areas
solo-nueva admin GET /api/v1/catastro/areas?q=AV-BAJA-AUTO 200 !AV-BAJA-AUTO
solo-nueva admin GET /api/v1/geo/areas 200 !AV-BAJA-AUTO
